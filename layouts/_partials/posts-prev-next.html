{{- /* 按发布时间排序的上一/下一篇文章导航。 取所有 Section=posts、非草稿页面，按 Date 倒序，找当前页索引取邻居。 */ -}} {{- $pages := where .Site.RegularPages "Section" "posts" -}} {{- $pages = where $pages "Params.draft" "ne" true -}} {{- $sorted := $pages.ByDate.Reverse -}} {{- $idx := -1 -}} {{- range $i, $p := $sorted -}} {{- if eq $p.Permalink $.Permalink -}} {{- $idx = $i -}} {{- end -}} {{- end -}} {{- $prev := "" -}} {{- if gt $idx 0 -}} {{- $prev = index $sorted (sub $idx 1) -}} {{- end -}} {{- $next := "" -}} {{- if and (ge $idx 0) (lt $idx (sub (len $sorted) 1)) -}} {{- $next = index $sorted (add $idx 1) -}} {{- end -}} {{- if or $prev $next -}}
{{- with $prev }} ← 上一篇 {{ .Title }} {{- end }} {{- with $next }} 下一篇 → {{ .Title }} {{- end }}
{{- end }}
